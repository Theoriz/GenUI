/*
The browser mirror of the in-game panel.

Every metric and colour comes from the --genui-* custom properties GenUIStyle.ToCss() emits, which
the schema message installs into the page: changing a token in Unity moves both interfaces. Only
what has no token of its own is written here, and each one says where its value comes from.
*/

:root {
  /* The panel's width, from the sample prefab's scroll view rather than from GenUIStyle. */
  --genui-panel-width: 323px;

  /* Space between panels: the scroll view content's layout spacing. */
  --genui-panel-spacing: 4.5px;

  /* The checkbox's box, ToggleUI's own 0.13 of the row beside the label half. */

  /* The 9-sliced sprites become rounded rectangles; see the deviations in the README. */
  --genui-corner-radius: 3px;

}
